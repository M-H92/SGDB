Variante 1

            CREATE TABLE logs(
	id serial primary key,
	log_user text not null,
	type_action text not null,
	date timestamp not null,
	old_value json,
	new_value json
);
 
CREATE OR REPLACE FUNCTION log_user_action()
RETURNS TRIGGER
AS
$$
BEGIN
	INSERT INTO logs(log_user,type_action,date,old_value,new_value) VALUES (
		(SELECT CURRENT_USER),
		TG_OP,
		(SELECT CURRENT_TIMESTAMP),
		(row_to_json(OLD)),
		(row_to_json(NEW))
	);
	IF TG_OP = 'DELETE' THEN RETURN OLD;
	ELSE RETURN NEW;
	END IF;
END
$$
LANGUAGE PLPGSQL;	
 
 
CREATE OR REPLACE TRIGGER on_update_nageur
	BEFORE UPDATE
ON nageur
	FOR EACH ROW EXECUTE PROCEDURE log_user_action();
 
CREATE OR REPLACE TRIGGER on_insert_nageur
	BEFORE INSERT
ON nageur
	FOR EACH ROW EXECUTE PROCEDURE log_user_action();
 
CREATE OR REPLACE TRIGGER on_delete_nageur
	BEFORE DELETE
ON nageur
	FOR EACH ROW EXECUTE PROCEDURE log_user_action();
 
 
DELETE FROM nageur where id=19;
INSERT INTO nageur(id,nom,prenom)values(19,'Carpenter','John');
UPDATE nageur SET prenom = 'Sabrina' where id=19;
SELECT * FROM nageur;
SELECT * FROM logs;

Variante 2

create or replace function t_log()
returns trigger
as
$$
begin
	insert into logs
		(nom_table, 
		type_action, 
		date_log, 
		old_value, 
		new_value, 
		comm)
	values
		(TG_TABLE_NAME, 
		TG_OP, 
		(select current_date), 
		concat(old.*), 
		concat(new.*), 
		TG_OP || ' in ' || TG_TABLE_NAME || 
			COALESCE(' from ' || old.*, '') || 
			COALESCE(' to ' || new.*, ''));
	IF TG_OP = 'DELETE'
	THEN
		RETURN OLD;
	END IF;
	RETURN NEW;
end;
$$ 
language plpgsql;

Variante 3

CREATE TABLE logs(
	id SERIAL PRIMARY KEY,
	log_date date,
	log_user text,
	log_table text,
	log_action text,
	old_value JSON,
	new_value JSON
);

CREATE OR REPLACE FUNCTION create_log()
RETURNS trigger
AS
$$
DECLARE
	old_val JSON = NULL;
	new_val JSON = NULL;
BEGIN

	IF TG_OP IN ('UPDATE', 'DELETE')
	THEN
		old_val := row_to_json(OLD);
	END IF;
	
	IF TG_OP IN ('INSERT', 'UPDATE')
	THEN
		new_val := row_to_json(NEW);
	END IF;
	
INSERT INTO logs
(
	log_date ,
	log_user ,
	log_table ,
	log_action ,
	old_value ,
	new_value 
) VALUES 
(
	(SELECT CURRENT_DATE),
	(SELECT CURRENT_USER),
	TG_TABLE_NAME,
	TG_OP,
	old_val,
	new_val
);

	RETURN NEW;
END;
$$
LANGUAGE PLPGSQL

CREATE OR REPLACE TRIGGER on_nageur
	AFTER INSERT OR UPDATE OR DELETE
ON nageur
	FOR EACH ROW
		EXECUTE PROCEDURE create_log();

SELECT * FROM nageur;
